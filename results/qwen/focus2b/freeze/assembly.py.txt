"""CPU-only assembly; reuse verified outcome-free inputs and historical receipts."""
from datetime import datetime, timezone
from pathlib import Path
from stencil import focus2 as f

root = Path('/home/bmarti44/stencil-llm')
folder = root / 'results/qwen/focus2b/freeze'
prior = f.parse_json((root / 'results/qwen/focus2/freeze/manifest.json').read_text())
manifest = f.parse_json((folder / 'manifest.json').read_text())
acceptance = Path('/tmp/focus2b-amendment1-acceptance.log').read_text()
f.require('90 passed, 1 xfailed' in acceptance, 'required CPU acceptance not passed')
(folder / 'acceptance.txt').write_text(acceptance)
(folder / 'assembly.py.txt').write_text(Path(__file__).read_text())
assets = f.parse_json((root / 'results/qwen/focus2/freeze/assets.json').read_text())
for role, old in assets.items():
    actual = f.asset_descriptor(root, old['path'], old['source_revision'])
    f.require(actual == old, 'asset changed: ' + role)
    assets[role] = actual
f.atomic_json(folder / 'assets.json', assets)
paths = {k: str((folder / d['path']).relative_to(root)) for k, d in manifest['files'].items()}
paths.update({k: str(p.relative_to(root)) for k, p in f.RUNTIME_SOURCES.items() if k not in f.ASSET_ROLES})
for role, descriptor in prior['files'].items():
    if role.startswith(('check', 'repair_code', 'development_evidence_', 'review_')) and role != 'checker_tests':
        f.member(root, 'HEAD', descriptor['path'], descriptor['sha256'])
        paths[role] = descriptor['path']
paths['focus2_diagnosis'] = 'results/focus2-competence-diagnosis-fable.md'
paths['focus2_prior_banks'] = 'results/qwen/focus2/freeze/banks.json'
for name in ('manifest.json', 'retired-inputs.json'):
    paths['amendment1_development_' + name.replace('.', '_')] = 'results/qwen/focus2b/development/' + name
(folder / 'ruff.txt').write_text(Path('/tmp/focus2b-amendment1-ruff.log').read_text())
paths['freeze_ruff'] = str((folder / 'ruff.txt').relative_to(root))
for name in ('assets.json', 'acceptance.txt', 'assembly.py.txt'):
    paths['freeze_' + name.replace('.', '_')] = str((folder / name).relative_to(root))
manifest['files'] = {k: {'path': p, 'sha256': f.sha((root / p).read_bytes())} for k, p in paths.items()}
manifest['files'].update(assets)
contents = {k: (b'' if k == 'model' else (root / d['path']).read_bytes()) for k, d in manifest['files'].items()}
f.verify_evidence(root, manifest, contents)
banks = f.parse_json(contents['banks'])
development = f.parse_json(contents['development'])
f.validate_banks(banks, development)
for label, source in development['sources'].items():
    f.member(root, 'HEAD', source['path'], source['sha256'])
rejections = [dict(episode=ep['id'], **rejection) for bank in banks.values() for ep in bank for rejection in ep['payload_rejections']]
f.require(sum(sum(ep['payload_attempts'].values()) for bank in banks.values() for ep in bank) == len(rejections), 'missing rejection receipt')
old_banks = f.parse_json(contents['focus2_prior_banks'])
old_fps = {f.fingerprint(ep['family'], payload) for bank in old_banks.values() for ep in bank for payload in ep['requests'].values()}
new_fps = {f.fingerprint(ep['family'], payload) for bank in banks.values() for ep in bank for payload in ep['requests'].values()}
f.require(not old_fps & new_fps, 'semantic collision with frozen FOCUS-2 inputs (no redraw)')
f.require(f.parse_json(contents['templates']) == f.template_manifest(f.load_tokenizer(root / assets['tokenizer']['path']), f.canonical(banks)), 'template drift')
for role, d in manifest['files'].items():
    if d.get('tracked', True) and not d['path'].startswith('results/qwen/focus2b/'):
        f.member(root, 'HEAD', d['path'], d['sha256'])
receipt = dict(
    status='VERIFIED', created_utc=datetime.now(timezone.utc).isoformat(), fit_on='none',
    acceptance_command="CUDA_VISIBLE_DEVICES='' uv run pytest -q tests/test_focus2.py tests/test_eval_data_separation.py tests/test_sealed_guard.py tests/test_no_side_effect_imports.py",
    acceptance=acceptance.strip().splitlines()[-1], ruff='All checks passed; touched source/CLI/tests',
    prepare_freeze='exit 0; CLI with committed outcome-free development manifest',
    bank_counts={k: len(v) for k, v in banks.items()}, request_payload_count=len(new_fps),
    development_unique_fingerprints=len(development['fingerprints']), collision_rejections=rejections,
    prior_focus2_unique_fingerprints=len(old_fps), prior_focus2_overlap=0,
    check39=dict(status='PROCEED_PLACEHOLDER', prereg_commit=f.PINS['prereg_commit'], receipt_commit=f.PINS['receipt_commit'], summary_sha256=f.PINS['summary_sha256'], gate_recomputed=True),
    check37='STOP unchanged, not pooled', section_sha256=f.sha(contents['section']),
    inherited_section_sha256=f.sha(contents['inherited_section']), no_model_outcomes=True,
    sealing_scope='Only the authorized tests/test_sealed_guard.py read sealed IFEval input; no other benchmark contact.',
)
f.atomic_json(folder / 'freeze-receipt.json', receipt)
manifest['files']['freeze_receipt'] = {'path': str((folder / 'freeze-receipt.json').relative_to(root)), 'sha256': f.sha((folder / 'freeze-receipt.json').read_bytes())}
review = dict(
    status='APPROVED', open_high_critical=0, section_sha256=f.sha(contents['section']),
    verification_basis="Brian's combined FOCUS-2b task authorizes direct registration verification as before. Verify the requested sort-only changes and inherited v3 dispositions against the required CPU suite; no claim that prior reviewers reran their reviews.",
    source_reviews=['results/focus2-review-fable.md', 'results/focus2-review-astra.md', 'results/focus2-competence-diagnosis-fable.md'],
    text_commit=f.PINS['focus2b_commit'], fix_commit=f.git_bytes(root, 'rev-parse', 'HEAD').decode().strip(),
    verified_changes=['sort-only 256 final/4 pilot/192 competence; three new seeds', '5-8 distinct -20..20 integers; reject both sorted forms; Amendment 1 deterministic collision rejection with attempt logs', 'explicit integer-array schema and unchanged-list default; obligations once in actual requests', '64 memo episodes; exact checker/repair/decisions/cost/one-attempt policy unchanged', 'amended and inherited immutable section bindings; fresh external asset hashes; complete exclusions and accepted-payload token maps', 'exact f86ae560 regression, within/across-bank rejection, rejection-log tamper refusal'],
    inherited_dispositions='FOCUS-2 v3 and its finding-named CPU tests carried forward; no reopened scientific gate or benchmark stage.',
    dependencies={k: v['sha256'] for k, v in manifest['files'].items()},
)
f.atomic_json(folder / 'review.json', review)
manifest['files']['review'] = {'path': str((folder / 'review.json').relative_to(root)), 'sha256': f.sha((folder / 'review.json').read_bytes())}
manifest.update(status='REGISTERED', candidate_only=False, remaining=[], output_path='results/qwen/focus2b/outputs')
(folder / 'manifest.json').write_text(f.canonical(manifest))
print(f.canonical({'status': 'FREEZE_READY', 'manifest_sha256': f.sha((folder / 'manifest.json').read_bytes()), 'dependencies': len(manifest['files']), 'receipt': receipt}))
