def acceptance_passed(text, exit_code):
    """Parse whole pytest result labels; xfailed is distinct from failed."""
    lines = [line.strip().strip("=").strip() for line in text.splitlines()]
    summaries = [line for line in lines if re.search(r"\b\d+ (?:passed|failed|xfailed|skipped|error|errors)\b.*\bin \d", line)]
    if not summaries:
        return False
    summary = summaries[-1].split(" in ", 1)[0]
    counts = dict((label, int(count)) for count, label in re.findall(r"(?:^|,\s*)(\d+) ([a-z]+)\b", summary))
    return exit_code == 0 and counts.get("failed", 0) == 0
