"""CPU-only FOCUS-2c Amendment 1 freeze assembly; no model construction."""
from datetime import datetime, timezone
from pathlib import Path
from stencil import focus2 as f

root = Path('/home/bmarti44/stencil-llm')
folder = root / 'results/qwen/focus2c/amendment1/freeze'
old_folder = root / 'results/qwen/focus2c/freeze'
prior = f.parse_json((old_folder / 'manifest.json').read_text())
manifest = f.parse_json((folder / 'manifest.json').read_text())
acceptance = Path('/tmp/focus2c-amendment1-acceptance.log').read_text()
f.require('97 passed, 1 xfailed' in acceptance and 'FAILED' not in acceptance, 'required CPU acceptance')
(folder / 'acceptance.txt').write_text(acceptance)
(folder / 'assembly.py.txt').write_text(Path(__file__).read_text())
(folder / 'ruff.txt').write_text(Path('/tmp/focus2c-amendment1-ruff.log').read_text())
assets = f.parse_json((old_folder / 'assets.json').read_text())
for role, old in assets.items():
    actual = f.asset_descriptor(root, old['path'], old['source_revision'])
    f.require(actual == old, 'asset changed: ' + role)
f.atomic_json(folder / 'assets.json', assets)
paths = {k: str((folder / d['path']).relative_to(root)) for k, d in manifest['files'].items()}
paths.update({k: str(p.relative_to(root)) for k, p in f.RUNTIME_SOURCES.items() if k not in f.ASSET_ROLES})
for role, descriptor in prior['files'].items():
    if role.startswith(('check', 'repair_code', 'development_evidence_', 'review_', 'excluded_inputs_')) and role != 'checker_tests':
        f.member(root, 'HEAD', descriptor['path'], descriptor['sha256'])
        paths[role] = descriptor['path']
paths['development_manifest_source'] = 'results/qwen/focus2c/development/manifest.json'
original_commit = 'ac3cae1'
original_commit = f.git_bytes(root, 'rev-parse', original_commit).decode().strip()
original_output = root / 'results/qwen/focus2c/outputs'
carry_paths = dict(manifest=old_folder / 'manifest.json', banks=old_folder / 'banks.json', config=old_folder / 'config.json')
for stage in ('competence', 'pilot'):
    for path in sorted((original_output / stage).rglob('*.json')):
        carry_paths[stage + '_' + path.stem] = path
carry = dict(commit=original_commit, output_path=str(original_output.relative_to(root)),
             pilot_episode='pilot:sort:ascending:0:0',
             files={role: dict(path=str(path.relative_to(root)), sha256=f.sha(path.read_bytes())) for role, path in carry_paths.items()})
f.atomic_json(folder / 'competence-carry.json', carry)
paths['competence_carry'] = str((folder / 'competence-carry.json').relative_to(root))
for name in ('assets.json', 'acceptance.txt', 'assembly.py.txt', 'ruff.txt'):
    paths['freeze_' + name.replace('.', '_')] = str((folder / name).relative_to(root))
manifest['files'] = {k: {'path': p, 'sha256': f.sha((root / p).read_bytes())} for k, p in paths.items()}
manifest['files'].update(assets)
contents = {k: (b'' if k == 'model' else (root / d['path']).read_bytes()) for k, d in manifest['files'].items()}
f.verify_evidence(root, manifest, contents)
banks = f.parse_json(contents['banks'])
f.require(banks == f.parse_json((old_folder / 'banks.json').read_text()), 'input bank changed')
development = f.parse_json(contents['development'])
f.validate_banks(banks, development)
tok = f.load_tokenizer(root / assets['tokenizer']['path'])
f.require(f.parse_json(contents['templates']) == f.template_manifest(tok, f.canonical(banks)), 'template drift')
old_templates = f.parse_json((old_folder / 'templates.json').read_text())
f.require([k for k, v in f.TEMPLATES.items() if f.sha(v) != old_templates['templates'][k]] == ['delay'], 'non-delay template changed')
for role, d in manifest['files'].items():
    if d.get('tracked', True) and not d['path'].startswith(str(folder.relative_to(root)) + '/'):
        f.member(root, 'HEAD', d['path'], d['sha256'])
pre = dict(root=root, banks=banks, tok=tok, binding={'hashes': {k:v['sha256'] for k,v in manifest['files'].items()}})
verified_carry = f.validate_carry(carry, pre)
f.require(verified_carry['spent'] == 467.0806055078283 and len(verified_carry['carried']) == 27, 'carry cost/records')
receipt = dict(status='VERIFIED', created_utc=datetime.now(timezone.utc).isoformat(), fit_on='none',
               acceptance=acceptance.strip().splitlines()[-1], ruff='clean',
               banks_unchanged=True, bank_counts={k: len(v) for k,v in banks.items()},
               changed_template_keys=['delay'], delay_user_tokens=len(f.encode(tok, f.delay_text(tok))),
               delay_caps=list(f.DELAY_CAPS), final_outcomes_observed=False,
               competence='original certificate PASS; 63/59/64; CPU revalidated, never rerun',
               original_competence_records=192, carried_pilot_records=27,
               prior_gpu_seconds=verified_carry['spent'], all_external_assets_freshly_hashed=True,
               original_manifest_sha256=f.sha((old_folder / 'manifest.json').read_bytes()),
               section_sha256=f.sha(contents['section']),
               inherited_section_sha256=f.sha(contents['inherited_section']))
f.atomic_json(folder / 'freeze-receipt.json', receipt)
manifest['files']['freeze_receipt'] = {'path': str((folder / 'freeze-receipt.json').relative_to(root)), 'sha256': f.sha((folder / 'freeze-receipt.json').read_bytes())}
review = dict(status='APPROVED', open_high_critical=0, section_sha256=f.sha(contents['section']),
              verification_basis="Brian's explicit FOCUS-2c Amendment 1 resume ruling; required CPU checks and direct verification. No claim of historical reviewers re-reviewing.",
              text_commit=f.PINS['candidate_commit'], fix_commit=f.git_bytes(root, 'rev-parse', 'HEAD').decode().strip(),
              verified_changes=['delay prompt/caps/retry/paired exclusions only; task law unchanged', 'original competence stands; raw records/checker/model verified', 'original completed unchanged pilot responses reused with audited provenance', 'same 6 GPU-hour budget includes old 467.0806055078283 seconds'],
              dependencies={k:v['sha256'] for k,v in manifest['files'].items()})
f.atomic_json(folder / 'review.json', review)
manifest['files']['review'] = {'path': str((folder / 'review.json').relative_to(root)), 'sha256': f.sha((folder / 'review.json').read_bytes())}
manifest.update(status='REGISTERED', candidate_only=False, remaining=[], output_path='results/qwen/focus2c/amendment1/outputs')
(folder / 'manifest.json').write_text(f.canonical(manifest))
print(f.canonical(dict(status='FREEZE_READY', manifest_sha256=f.sha((folder/'manifest.json').read_bytes()), receipt=receipt)))
