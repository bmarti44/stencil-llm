"""CPU-only report and additional artifact audit; no generation or fitting."""
import hashlib
import json
import sys
from collections import Counter
from pathlib import Path

ROOT = Path('/home/bmarti44/stencil-llm')
sys.path.insert(0, str(ROOT / 'scripts'))
import focus_check40d as check
import torch

OUT = check.OUT
summary = json.loads((OUT / 'summary.json').read_text())
rows = [json.loads(line) for line in (OUT / 'records.jsonl').read_text().splitlines()]
tasks = json.loads((OUT / 'tasks.json').read_text())
biases = torch.load(OUT / 'biases.pt', map_location='cpu', weights_only=True)
prior = torch.load(check.OLD / 'frozen-biases.pt', map_location='cpu', weights_only=True)
assert torch.equal(biases['js'], prior['correct'] * .75)
assert torch.equal(biases['python'], prior['swapped'] * .75)
assert torch.equal(biases['js'] * (2/3), prior['correct'] * .5)
assert torch.equal(biases['python'] * (2/3), prior['swapped'] * .5)
generator = torch.Generator().manual_seed(check.SEED)
for episode in biases['shuffled']:
    for step in episode:
        expected = torch.stack([row[torch.randperm(len(row), generator=generator)] for row in biases['js']])
        assert torch.equal(expected, step)
        assert torch.allclose(step.norm(dim=-1), biases['js'].norm(dim=-1))
assert tasks == check.make_tasks(len(tasks))
previous_expressions = {t['expression'] for bank in check.base.expression_bank().values() for t in bank}
assert previous_expressions.isdisjoint(t['expression'] for episode in tasks for t in episode.values())
assert len({t['expression'] for e in tasks for t in e.values()}) == len(tasks)*5
for row in rows:
    ids = row['generated_token_ids']
    assert 0 < len(ids) <= 64 and row['token_cap'] == 64
    assert row['truncated'] == (not row['eos'] and len(ids) == 64)
    assert not row['cost_stopped'] if summary['complete'] else True
    prefix = row['cache_prefix_token_ids']
    assert row['cache_prefix_sha256'] == hashlib.sha256(json.dumps(prefix).encode()).hexdigest()
    assert row['input_sha256'] == hashlib.sha256(json.dumps(row['input_token_ids']).encode()).hexdigest()
    assert row['cue'] == check.cue_for(row['arm'], row['step'])
    assert row['bias_active'] == (row['bias_sha256'] is not None)
    if row['arm'] == 'fresh-OFF':
        assert len(row['history']) == 2 and not row['retained_kv']
    else:
        assert len(row['history']) == 2 + 2 * check.STEPS.index(row['step'])
    if row['eos']:
        assert not row['appended_terminal_token_ids']
    else:
        assert len(row['appended_terminal_token_ids']) == int(row['retained_kv'])
    assert [m['role'] for m in row['history']] == ['system'] + ['user', 'assistant'] * ((len(row['history'])-2)//2) + ['user']
assert json.loads((OUT / 'runtime.json').read_text())['raw_slot_verified_layers'] == 48
kernel = json.loads((OUT / 'kernel.json').read_text())
assert kernel['adopted'] and kernel['exact_off_next_logits'] and kernel['nonzero_dispatch_verified']
audit = json.loads((OUT / 'audit.json').read_text())
audit.update(frozen_biases_rederived=True, every_shuffled_permutation_rederived=True,
             task_bank_and_lineage_rederived=True, token_caps_pair_closures_hashes=True,
             real_48_layer_raw_slot_and_exact_off=True)
check.write('audit.json', audit)
